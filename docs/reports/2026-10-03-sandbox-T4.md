import { spawn, execFileSync } from "node:child_process";
import { writeFileSync } from "node:fs";
import { join } from "node:path";

const [mountRoot, outDir, prefix, , ...serverArgv] = process.argv.slice(2);
const IMAGE = "mcpeak-audit-sandbox:9199c91b63ce";
const NAME = "mcpeak-audit-fixture-target";
const VOLUME = "mcpeak-audit-fixture-trace";
const HOME = new URL("./home", import.meta.url).pathname;
const TRACE = "trace=openat,execve,connect,?unlink,unlinkat,?rmdir,?rename,?renameat,renameat2";

execFileSync("docker", ["volume", "create", VOLUME]);
const child = spawn("docker", [
  "run", "-i", "--name", NAME, "--read-only", "--cap-drop", "ALL",
  "--security-opt", "no-new-privileges", "--pids-limit", "256", "--memory", "1g",
  "--tmpfs", "/tmp:rw,size=256m,uid=1000", "--hostname", "jiwoo-laptop", "--network", "none",

  "-w", "/workspace", IMAGE,
  "strace", "-f", "-qq", "-s", "256", "-e", TRACE, "-o", "/var/mcpeak/trace", ...serverArgv,
], { stdio: ["pipe", "pipe", "inherit"] });

let buffer = "";
const waiters = new Map();
child.stdout.on("data", (chunk) => {
  buffer += chunk;
  let index;
  while ((index = buffer.indexOf("\n")) >= 0) {
    const line = buffer.slice(0, index);
    buffer = buffer.slice(index + 1);
    if (line.trim() === "") continue;
    const message = JSON.parse(line);
    waiters.get(message.id)?.(message);
  }
});
let nextId = 1;
const request = (method, params) => new Promise((resolve) => {
  const id = nextId++;
  waiters.set(id, resolve);
  child.stdin.write(`${JSON.stringify({ jsonrpc: "2.0", id, method, params })}\n`);
});
const sleep = (ms) => new Promise((r) => setTimeout(r, ms));
let reader = 0;
let seen = 0;
/** 직전 mark 이후 새로 생긴 줄을 그 단계 이름의 파일로 쓴다(§3.2 의 mark 와 같은 방법). */
async function mark(phase) {
  await sleep(300);
  const all = execFileSync("docker", ["run", "--rm", "--name", `mcpeak-audit-fixture-reader-${reader++}`,
    "--network", "none", "-v", `${VOLUME}:/v:ro`, IMAGE, "cat", "/v/trace"], { encoding: "utf8", maxBuffer: 1 << 28 });
  const lines = all.split("\n").filter((line) => line !== "");
  writeFileSync(join(outDir, `${prefix}-${phase}.txt`), lines.slice(seen).map((l) => `${l}\n`).join(""));
  console.error(`${phase}: ${lines.length - seen} lines`);
  seen = lines.length;
}

const init = await request("initialize", { protocolVersion: "2025-06-18", capabilities: {}, clientInfo: { name: "mcpeak-fixture", version: "0.0.0" } });
child.stdin.write(`${JSON.stringify({ jsonrpc: "2.0", method: "notifications/initialized" })}\n`);
console.error("server:", init.result?.serverInfo?.name);
await mark("start");
const list = await request("tools/list", {});
writeFileSync(join(outDir, `${prefix}-tools.json`), `${JSON.stringify(list.result.tools, null, 2)}\n`);
await mark("list");
const calls = JSON.parse(process.env.CALLS ?? "[]");
for (const [i, call] of calls.entries()) {
  const result = await request("tools/call", { name: call.name, arguments: call.args });
  console.error("call", call.name, JSON.stringify(result.result ?? result.error).slice(0, 120));
  await mark(`call${i}-${call.name}`);
}
await request("tools/list", {});
await mark("list2");
child.stdin.end();
await new Promise((resolve) => child.on("exit", resolve));
await mark("shutdown");
execFileSync("docker", ["rm", "-f", NAME]);
execFileSync("docker", ["volume", "rm", VOLUME]);

### 떠 오지 못한 것

- **x86_64 에만 있는 `unlink`·`rmdir`·`rename`.** `docker build --platform linux/amd64` 로 같은 Dockerfile 을
  빌드해 에뮬레이션에서 탐침을 돌렸지만 strace 가 `[ Process PID=13 runs in x32 mode. ]` 와
  `syscall_0x6bf8650(…)` 같은 깨진 줄 7099줄만 냈다. 쓸 수 없어 버렸고 이미지도 지웠다. 그래서 이 세 이름의
  테스트 줄은 **채집한 것이 아니다.** arm64 에서 뜬 `unlinkat`·`renameat` 줄에서 dirfd 와 flags 인자를 뺀
  꼴이고, 테스트 주석에 그렇게 적었다. x86_64 의 실물은 CI 의 E2E(T6)가 처음 본다.
- **게이트웨이로 가는 접속.** 게이트웨이가 아직 없다.

## 잡음 표 (`NOISE_TABLE`, 여덟 행)

| 행 | 빼는 것 | 근거 |
|---|---|---|
| `first-exec` | `start` 단계의 첫 `exec` | strace 가 띄운 서버 명령 자체다. `trace-weather-start.txt` 첫 줄 |
| `package-runner` | 명령이 `npx`·`npm` 일 때 `start` 단계의 `exec` 중 basename 이 `node`·`sh`·`npm`·`npx` | `trace-npx-start.txt` 에서 npx 가 `node /usr/local/bin/npx`, `sh -c "tsc --version"`, `node …/typescript/bin/tsc` 를 차례로 실행한다 |
| `package-bin` | 같은 조건에서 경로에 `/node_modules/.bin/` 이 든 `exec` | 같은 파일의 `execve("/workspace/node_modules/.bin/tsc", …)`. 실행기가 고른 패키지의 bin 이다 |
| `bin-shim-helper` | 같은 조건에서 basename 이 `sed`·`dirname`·`uname` | 같은 파일에서 pnpm 의 bin 셸 스크립트가 자기 위치를 구하느라 셋을 실행한다 |
| `dev-null` | 쓰기 열기 중 경로가 정확히 `/dev/null` | `trace-split.txt` 에서 `sh` 가 `> /dev/null` 리다이렉트로 `O_WRONLY\|O_CREAT\|O_TRUNC` 로 연다 |
| `unix-socket` | `connect` 중 `family` 가 `unix` | `trace-connect.txt`·`trace-exec.txt` 에서 glibc 가 `/var/run/nscd/socket` 을 두드린다 |
| `embedded-resolver` | `connect` 중 `inet` `127.0.0.11` 의 53번 | `trace-embedded-dns.txt`. 사용자 정의 네트워크에서 `/etc/resolv.conf` 의 nameserver 가 이 주소다 |
| `gateway` | `connect` 중 주소가 `<gateway>` 이고 포트가 53·80·443 | **채집 근거 없음.** §3.4 표와 T5 의 포트 문장 계약으로 두었다 |

§3.4 의 다섯 행과 다른 점.

- **더한 것 셋.** `package-bin`, `bin-shim-helper`, `embedded-resolver`. 셋 다 채집한 기록에 있었고, 없으면
  각각 npx 로 띄운 정상 서버에 `child-process` 가, 이름을 한 번 묻는 정상 서버에 `internal-address` 가 난다.
- **좁힌 것 하나.** §3.4 는 쓰기 열기의 `/dev/null`, `/dev/tty`, `/dev/pts/` 아래, `/proc/self/` 아래를 뺀다.
  채집한 네 기록(날씨 서버, 탐침, mv, npx)에서 `/tmp` 밖 쓰기 열기로 나온 런타임 경로는 `/dev/null` 하나였다.
  나머지 셋은 **넣지 않았다.** Node 가 표준 입출력을 다루면서 그 경로를 쓰기로 여는 일은 이 채집에서 한 번도
  없었고, `/proc/self/` 를 접두로 빼면 서버가 `/proc/self/mem` 이나 `oom_score_adj` 에 쓰는 것도 함께 사라진다.
  TTY 를 붙이지 않는 실행(`-i` 만)이라 `/dev/tty`·`/dev/pts/` 가 나올 길도 없어 보인다.
- **근거가 약한 것 하나.** `dev-null` 의 근거는 탐침이 스스로 쓴 `sh -c "cat /etc/hostname > /dev/null"` 이다.
  런타임이 저절로 연 것이 아니다. 다만 `/dev/null` 에 쓰는 것은 어디에도 남지 않아 "임시 디렉터리 밖 쓰기" 로
  볼 이유가 없고, 셸을 띄우는 서버에서 흔히 나온다. 정확 일치로만 뺀다.
- 읽기 전용 열기는 표에 없다. 규칙이 좁아서 애초에 닿지 않는다(§3.4 의 설명 그대로).
- `filterNoise` 의 `context.mountRoot` 는 쓰지 않는다. 표의 어느 행도 마운트 루트를 보지 않는다.

## 다른 태스크가 알아야 할 것

1. **이름 조회는 게이트웨이 주소가 아니라 `127.0.0.11:53` 으로 `connect` 한다.** `--internal` 네트워크에
   `--dns <게이트웨이 IP>` 로 띄워 확인했다. 컨테이너의 `/etc/resolv.conf` 는 `nameserver 127.0.0.11` 이고
   (`# ExtServers: [10.9.9.9]`), Docker 의 내장 리졸버가 `--dns` 로 넘겨 준다. 그래서 백엔드(T1)가 게이트웨이
   IP 를 `<gateway>` 로 바꿔도 이름 조회의 `connect` 는 바뀌지 않는다. `127.0.0.0/8` 이라 그대로 두면 이름을
   묻는 모든 서버에 `behavior/internal-address` 가 난다. 잡음 표의 `embedded-resolver` 행이 뺀다. T3 의
   `network/direct-ip` 는 내부 주소 표 안의 주소를 보지 않으므로 영향이 없을 것이나 확인하지 않았다.
2. **단계 경계에 걸친 갈라진 줄.** `parseTrace` 는 상태가 없다(시그니처가 그렇다). `<unfinished ...>` 가 한
   `mark` 의 끝에, `<... resumed>` 가 다음 `mark` 의 처음에 오면 양쪽에서 한 줄씩 `unparsed` 로 센다. 그러면
   `skipped` 의 "n줄을 해석하지 못했습니다" 가 실행마다 달라질 수 있다. T1 이 끝의 `<unfinished ...>` 줄을
   다음 `mark` 로 미루면 없어진다.
3. **`exfiltrated` 의 `name` 은 `~/<HomeFile.path>` 꼴이어야 한다.** `runBehaviorRules` 가 그 글자로 대조한다
   (`CanaryValue.name` 의 주석과 같다). 취약 예제의 `sync_settings` 는 이것이 넘어오지 않으면
   `file-canary-read` 를 한 건 더 낸다(§1.3 의 "각각 정확히 한 번" 이 깨진다).
4. **npx 로 띄우는 서버.** 채집에서 npm 이 `~/.npm/_logs/…` 와 `~/.npm/_update-notifier-last-checked` 를 쓰기로
   열다 `ENOENT` 로 실패했다(`missing` 이라 발견은 아니다). 홈이 읽기 전용이라 npm 캐시를 만들 수 없다는
   뜻이다. `--no-install` 로 로컬 bin 을 띄우는 것은 됐고, 레지스트리에서 받는 실행은 해 보지 않았다.

## 임의로 판단한 부분

규칙(`behavior.ts`).

1. **접힌 발견의 문장은 가장 심각한 관측의 것이다.** §3.4 는 심각도(최댓값)와 위치(첫 호출)만 정한다.
   `evidence` 첫 칸은 정렬한 대상의 첫째라, 문장의 대상과 첫 칸이 다를 수 있다(예: 문장은 `169.254.169.254:80`,
   첫 칸은 `10.0.0.1:443`). high 발견이 medium 짜리 대상을 말하면 읽는 사람이 속는다고 봤다. 같은 심각도
   안에서는 대상의 코드 단위 순이다.
2. **`annotation-violation` 의 (b)·(d)와 `write-outside` 의 지우기·옮기기는 `result` 가 `ok`·`denied` 일 때만.**
   표의 (b)·(d)에는 `result` 조건이 없다. (a)와 `write-outside` 에 맞췄다. 없는 파일을 지우려 한 것(`missing`)과
   `EXDEV` 같은 것(`error`)은 내지 않는다.
3. **`annotation-violation` 의 (c)와 `child-process` 는 성공한 `exec` 만.** PATH 를 훑다 실패한 줄을 세지 않는다.
4. **`destructiveHint` 가 false 인 것은 지우기·옮기기에만 건다.** 표 그대로다. `readOnlyHint` 도 true 면
   문장은 `readOnlyHint` 쪽이다.
5. **`/var/mcpeak` 아래의 지우기·옮기기도 `observation-tampered` 로 본다.** 표는 쓰기 열기와 `traceShrank` 만
   적었다. 기록 파일을 지우는 것이 가장 뻔한 훼손이라 넣었다. 그리고 `/var/mcpeak` 아래의 쓰기·지우기는
   `write-outside` 로 따로 내지 않는다(훼손 발견이 더 무겁게 말한다). `annotation-violation` 은 함께 난다.
6. **`traceShrank` 는 훼손 사건이 하나도 없을 때만 한 건을 더한다.** "쓰기 열기가 있으면 한 번, traceShrank 가
   true 여도 한 번" 을 "둘이 겹쳐도 한 건" 으로 읽었다. 이 경우 위치는 서버 전체, `evidence` 는
   `["/var/mcpeak/trace", "시스템 콜 기록의 줄 수가 줄었습니다"]` 다(언제 줄었는지 모르므로 단계를 적지 않는다).
7. **서버 단계의 관측이 여러 단계에 걸치면 `단계: 시작, 종료` 처럼 잇는다.** §6.2 는 하나짜리 꼴만 적었다.
8. **IPv6 는 `[::1]:8080` 으로 적는다.** §6.2 의 `<ip>:<port>` 에 콜론이 겹쳐서다. IPv4 를 담은 IPv6
   (`::ffff:127.0.0.1`)는 그 IPv4 로 내부 주소 표에 댄다.
9. **경로·argv·callId 는 `escapeInvisible` 을 거쳐 문장과 `evidence` 에 싣는다.** 서버가 고른 글자가 터미널을
   조작하지 못하게 하는 기존 규약(`report.ts` 의 주석)을 따랐다. argv 는 80자로 자른 뒤에 바꾼다.
10. **응답 대조는 `location` 에 `toolIndex` 가 있는 문자열 전부.** `collectResultStrings` 는 `kind: "result"` 만
    낸다.
11. `BehaviorContext.mountRoot` 와 `calls[].outcome` 은 쓰지 않는다. §3.4·§6.2 에 그것을 보는 조건이 없다.
12. 결과 순서는 서버 전체, 도구 번호, 규칙 id 순이다. 최종 정렬은 `finalize` 의 몫이다.

해석기(`trace.ts`).

13. **절대 경로의 `.`·`..` 과 겹친 `/` 도 접는다.** §8.5 의 "절대 경로는 cwd 와 무관하게 그대로다" 는 cwd 와
    무관하다는 뜻으로 읽었다. 서버가 `NOTES + "/" + path` 로 이어 붙이면 `/home/node/notes/../.aws/credentials` 가
    그대로 찍히고, 접지 않으면 미끼 대조에 걸리지 않는다.
14. **strace 는 경로를 자르지 않는다.** `-s` 는 문자열에만 적용되고 파일 이름은 문자열로 치지 않는다
    (`trace-longpath.txt` 의 367자 경로가 온전하다). 잘리는 것은 `execve` 의 argv 다. §8.5 의 "잘린 경로" 스펙은
    잘린 `execve` 줄로 채웠다. 그 줄은 사건이 되지 않고 `unparsed` 로 간다. 256자를 넘는 인자로 프로세스를 띄운
    서버는 `child-process` 대신 `skipped` 의 줄 수로만 드러난다.
15. **dirfd 가 `AT_FDCWD` 가 아닌 상대 경로는 `unparsed`.** 그 fd 가 어느 디렉터리인지 한 줄로는 모른다.
    `du` 가 디렉터리를 내려가거나 `rm -r` 이 돌면 이런 줄이 나온다. 채집에서는 한 줄도 없었다.
16. 신호 줄(`--- SIGCHLD … ---`)과 종료 줄(`+++ … +++`)은 사건도 `unparsed` 도 아니다. `-qq` 에서도 신호 줄은
    찍힌다(채집에서 봤다). `sa_family` 가 `AF_INET`·`AF_INET6`·`AF_UNIX` 가 아닌 `connect` 는 `unparsed` 다.
17. 추적 목록에 없는 `open` 도 받는다(`openat` 과 같은 사건). 다른 strace 설정에서 온 줄을 버리지 않으려는 것이다.
18. 추상 unix 소켓은 `@이름` 으로 적는다. 채집에는 없었다.

호출 계획과 격리 홈.

19. **`format` 이 `uri`·`url` 이면 이름이 경로여도 URL 로 분류한다.** §8.5 의 "이름과 무관하게" 를 그렇게 읽었다.
20. `type` 이 `["string", "null"]` 같은 배열이어도 문자열 속성으로 본다. `$ref` 는 풀지 않는다.
21. 상한 문장의 도구 이름은 표기로 바꾸지 않았다. 단계 1 의 `인자를 만들 수 없어 호출하지 않았습니다` 문장이
    그렇게 한다(`audit.ts`).
22. **`planHome` 은 `random()` 이 낸 값이 한 번이라도 되풀이되면 던진다.** 카나리 값이 겹칠 때만 던지는 것보다
    엄격하다. §8.5 의 "random 이 같은 값을 두 번 내면 던진다" 를 글자대로 읽었다.
23. `.cursor/mcp.json` 의 가짜 서버 이름은 `internal-api`, 명령은 `npx -y internal-api-mcp` 다. 파일 내용은 전부
    줄바꿈으로 끝난다.

픽스처와 테스트.

24. `behavior-*.json` 의 키는 §8.1 의 것에서 `declaredHosts`·`canaries`·`mode` 를 빼고(`network` 가족의 것이다)
    `home`·`resultStrings`·`exfiltrated`·`mountRoot`·`command` 를 더했다. 반례는 `.counter.json` 이다.
25. 내부 주소 표, 실행기 표, §3.3 의 토큰 표, §3.5 의 파일 표는 파일 픽스처가 아니라 `it.each` 의 행으로 걸리는
    것과 반례를 두었다.
26. `NOISE_TABLE` 을 `noise.ts` 에서 export 했다(`index.ts` 에는 없다). 테스트가 행 이름과 근거를 본다.

## 남은 위험

- x86_64 전용 시스템 콜 이름 셋은 실물을 보지 못했다(위).
- `gateway` 행은 채집 근거가 없고, `dev-null` 행의 근거는 탐침이 만든 것이다(위).
- `/dev/tty`·`/dev/pts/`·`/proc/self/` 를 넣지 않았다. 실제 서버에서 런타임이 그 경로를 쓰기로 열면
  `write-outside` 오탐이 난다. T6 의 E2E 다섯 서버가 처음으로 가린다.
- 잡음 표의 `package-runner`·`package-bin`·`bin-shim-helper` 는 npx 가 pnpm 워크스페이스의 로컬 bin 을 띄운 한 번의
  관측에서 왔다. npm 이 만든 bin(심볼릭 링크)이나 레지스트리에서 받는 실행은 자식이 다를 수 있다.
- `first-exec` 는 사건의 순서에 기댄다(첫 줄). 규칙은 순서에 기대지 않지만 이 한 행은 그렇다. 백엔드가 사건을
  기록 순서대로 넘겨야 한다.
- 서버가 작업 디렉터리를 바꾸면 상대 경로가 틀어지고, 심볼릭 링크는 풀지 않는다(§3.4 가 한계로 둔 것).
- 유닛테스트는 인메모리와 픽스처만 본다. 위 스크래치 실행이 한 번의 실물 확인이고, 게이트웨이와 실제 백엔드를
  거친 확인은 T6 의 몫이다.
